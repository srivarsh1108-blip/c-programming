int main()
{
    int no,flag=0; 
    printf("Enter No : ");
    scanf("%d",&no); 
    for(int i=2;i<no;i++)
     {
        if(no%i==0)
        {
            flag=1;
            break; // come out from for loop
        }
    }
    if(flag==0)
      printf("\n%d is prime no  ",no);
    else
      printf("\n%d is not prime no  ",no);

// Sum of Even numbers and sum of odd number using single for loop

#include <stdio.h>
int main()
{
    int no,i,j,evensum=0,oddsum=0; 
    printf("Enter   No : ");
    scanf("%d",&no);  // 10
    for( i=1, j=2;i<=no && j<=no;i=i+2,j=j+2)
    { 
        evensum=evensum+j;
        oddsum=oddsum+i;
     }
     printf("\nEven Sum : %d",evensum);
     printf("\nOdd Sum : %d",oddsum);
    return 0;
}

Armstrong Number

1 5 3
3*3*3	= 27
5 * 5*5	= 125
1 * 1*1	=     1
               ---------------
		   153
               ---------------

#include <stdio.h>
int main()
{
    int no,sum=0,a,m; 
    printf("Enter   No : ");
    scanf("%d",&no);   // no - 153
    m=no; // m - 153
    while(no>0)
    {
        a=no%10;  // a - 3 , 5 , 1
        sum=sum+(a*a*a); // sum - 27 , 152 , 153
        no=no/10; // no -15 , 1 , <0
    }
    if(m==sum)
            printf("Armstrong Number");
    else
            printf("Not armstrong number");
    
     return 0;
}

#include <stdio.h>
int main()
{
    int no,a=1; 
    printf("Enter  no : ");
    scanf("%d",&no);  // 5
    for(int i=1;i<=no;i++) // outer for
    {
        for(int j=1;j<=i;j++) // inner for
        {
            printf(" %d",a++);
        } // j++
        printf("\n");
    } // i++
    return 0;
}
