      for(int j=i+1;j<size;j++)

int main()
{
    int size;
    printf("Enter  size : ");
    scanf("%d",&size);  // 7
    int arr1[size][size];
    int primarysum=0, secondsum=0;
    for(int i=0;i<size;i++) // outer for - row  
    { 
        for(int j=0;j<size;j++) // inner for - col 
        {
        printf("Enter Value in %d row %d col : " , i,j);
      scanf("%d",&arr1[i][j]);
        }  // i++
    } 
for(int i=0;i<size;i++) // outer for - row  
    { 
primarysum=primarysum+arr1[i][i];
secondsum=secondsum+arr1[i][size-i-1];
    }
